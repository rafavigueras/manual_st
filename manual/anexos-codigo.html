Los programas que producen todas las cifras del manual, con su salida.

Todas las cifras de los bloques NÚMEROS, PALANCA y CASO JST que no salen directamente de una fuente pública salen de estos programas, y la salida que aparece debajo de cada uno es la de su última ejecución, generada automáticamente el 17/10/2026 por herramientas/generar_anexos_codigo.py. Están escritos en Python con numpy, pandas, scipy y statsmodels, y se ejecutan desde la raíz del proyecto con el entorno virtual del repositorio. """Anexo S0: un stress test de credito de juguete, de principio a fin.

Una filial mexicana con dos carteras (consumo y empresas), el escenario base y
adverso de la JERS para Mexico del ejercicio 2025, un satelite por cartera,
flujos entre stages, provisiones IFRS 9 con prevision perfecta, RWA estandar y
caida de CET1 atribuible al credito. Balance estatico: lo que vence se repone,
el S3 no se cura ni se da de baja.

Simplificaciones deliberadas (se explican en el capitulo 8):
  - solo credito: no hay margen de intereses, comisiones ni gastos;
  - satelites dados (el de consumo es el estimado en el anexo S1);
  - transiciones a S2 y desde S2 como multiplos fijos de la PD de S1;
  - sin descuento financiero en la perdida a vida.
"""
import numpy as np
import pandas as pd

# ------------------------------------------------------------------
# 1. Escenario JERS 2025 para Mexico (tabla 4.1.1 y 4.1.2 del escenario)
# ------------------------------------------------------------------
ANOS = [2025, 2026, 2027]
PARO_2024 = 3.0
ESC = {
    "base":    {"pib": [1.3, 2.0, 2.3],   "paro": [3.3, 3.4, 3.5]},
    "adverso": {"pib": [-4.7, -2.2, -0.1], "paro": [5.2, 7.6, 8.8]},
}

# ------------------------------------------------------------------
# 2. Las dos carteras (millones de euros) y sus parametros
# ------------------------------------------------------------------
CARTERAS = {
    "consumo": {
        "stock": {"S1": 880.0, "S2": 70.0, "S3": 50.0},
        "satelite": (-3.514, 0.215, -0.0802),   # anexo S1: logit PD = a + b*paro(t-1) + c*PIB(t)
        "m_s1_s2": 1.5,   # tasa S1->S2 = m * PD S1
        "m_s2_s3": 3.0,   # PD S2 = m * PD S1
        "cura_s2": 0.20,  # S2 -> S1 cada ano
        "vida": 3,        # anos de vida residual para la perdida a vida de S2
        "lgd": lambda paro, pib: min(0.85, 0.70 + 0.01 * (paro - 3.5)),
        "pond_rwa": 0.75,
        "margen": {"base": 0.085, "adverso": 0.075},   # margen antes de provisiones, dado
    },
    "empresas": {
        "stock": {"S1": 950.0, "S2": 36.0, "S3": 14.0},
        "satelite": (-4.60, 0.10, -0.12),
        "m_s1_s2": 2.0,
        "m_s2_s3": 6.0,
        "cura_s2": 0.20,
        "vida": 4,
        "lgd": lambda paro, pib: 0.40 + 0.02 * max(0.0, -pib),
        "pond_rwa": 1.00,
        "margen": {"base": 0.015, "adverso": 0.012},
    },
}
TIPO_FISCAL = 0.30
CET1_INICIAL_SOBRE_RWA = 0.13


def logistica(z):
    return 1 / (1 + np.exp(-z))


def sendas(cart, esc):
    """PD de S1 y LGD por ano: 2024 (partida), 2025-2027 y 10 anos mas tras el horizonte.

    Tras 2027: en el base, parametros planos; en el adverso, reversion lineal en
    6 anos a los parametros base de 2027 (como en la metodologia EBA 2027).
    """
    a, b, c = cart["satelite"]
    paro_prev = [PARO_2024] + ESC[esc]["paro"][:-1]
    pd_h = [logistica(a + b * pp + c * g) for pp, g in zip(paro_prev, ESC[esc]["pib"])]
    lgd_h = [cart["lgd"](p, g) for p, g in zip(ESC[esc]["paro"], ESC[esc]["pib"])]
    # punto de partida (fin de 2024): parametros "de hoy" = primer ano del base
    pd0 = logistica(a + b * PARO_2024 + c * ESC["base"]["pib"][0])
    lgd0 = cart["lgd"](ESC["base"]["paro"][0], ESC["base"]["pib"][0])
    # base de 2027 como ancla de la reversion
    pb = [PARO_2024] + ESC["base"]["paro"][:-1]
    pd_b27 = logistica(a + b * pb[2] + c * ESC["base"]["pib"][2])
    lgd_b27 = cart["lgd"](ESC["base"]["paro"][2], ESC["base"]["pib"][2])
    pd_tras, lgd_tras = [], []
    for k in range(1, 11):
        w = max(0.0, 1 - k / 6) if esc == "adverso" else 1.0
        pd_tras.append(pd_b27 + (pd_h[2] - pd_b27) * w if esc == "adverso" else pd_h[2])
        lgd_tras.append(lgd_b27 + (lgd_h[2] - lgd_b27) * w if esc == "adverso" else lgd_h[2])
    return [pd0] + pd_h + pd_tras, [lgd0] + lgd_h + lgd_tras


def provisiones(cart, stock, pd_s, lgd_s, t, prevision_perfecta=True, prov_s3_vieja=None):
    """Stock de provisiones al final del ano t (t=0 es la partida)."""
    s1 = stock["S1"] * pd_s[t + 1] * lgd_s[t + 1]
    if prevision_perfecta:
        camino = [min(0.6, cart["m_s2_s3"] * pd_s[t + j]) for j in range(1, cart["vida"] + 1)]
    else:
        camino = [min(0.6, cart["m_s2_s3"] * pd_s[t + 1])] * cart["vida"]
    pd_vida = 1 - np.prod([1 - p for p in camino])
    s2 = stock["S2"] * pd_vida * lgd_s[t + 1]
    s3 = stock["S3_nuevo"] * lgd_s[max(t, 1)] + prov_s3_vieja
    return s1 + s2 + s3, {"S1": s1, "S2": s2, "S3": s3}


def proyectar(nombre, esc, prevision_perfecta=True):
    cart = CARTERAS[nombre]
    pd_s, lgd_s = sendas(cart, esc)
    pd_b, lgd_b = sendas(cart, "base")
    st = dict(cart["stock"])
    st["S3_nuevo"] = 0.0
    exp_total = st["S1"] + st["S2"] + st["S3"]
    # el punto de partida es un dato: se provisiona con el escenario interno (el base),
    # igual en los dos escenarios
    prov_s3_0 = st["S3"] * lgd_b[0]
    prov0, _ = provisiones(cart, st, pd_b, lgd_b, 0, True, prov_s3_0)
    filas = [{"ano": 2024, "S1": st["S1"], "S2": st["S2"], "S3": st["S3"], "prov": prov0,
              "deterioro": 0.0, "prov_S3": prov_s3_0}]
    prov_ant = prov0
    prov_s3_vieja = max(prov_s3_0, st["S3"] * lgd_s[1])   # suelo del S3 antiguo
    for t in (1, 2, 3):
        pd1 = pd_s[t]
        pd2 = min(0.6, cart["m_s2_s3"] * pd1)
        a_s3_desde_s1 = st["S1"] * pd1
        a_s2_desde_s1 = st["S1"] * cart["m_s1_s2"] * pd1
        a_s3_desde_s2 = st["S2"] * pd2
        a_s1_desde_s2 = st["S2"] * cart["cura_s2"]
        st["S1"] += -a_s3_desde_s1 - a_s2_desde_s1 + a_s1_desde_s2
        st["S2"] += a_s2_desde_s1 - a_s3_desde_s2 - a_s1_desde_s2
        st["S3_nuevo"] += a_s3_desde_s1 + a_s3_desde_s2
        st["S3"] = cart["stock"]["S3"] + st["S3_nuevo"]
        prov, det = provisiones(cart, st, pd_s, lgd_s, t, prevision_perfecta, prov_s3_vieja)
        filas.append({"ano": ANOS[t - 1], "S1": st["S1"], "S2": st["S2"], "S3": st["S3"],
                      "prov": prov, "deterioro": prov - prov_ant, "pd_s1": pd1,
                      "prov_S1": det["S1"], "prov_S2": det["S2"], "prov_S3": det["S3"]})
        prov_ant = prov
    df = pd.DataFrame(filas)
    # RWA estandar: performing a su ponderacion; S3 al 100 % neto de su provision
    df["rwa"] = (df.S1 + df.S2) * cart["pond_rwa"] + (df.S3 - df.prov_S3).clip(lower=0) * 1.00
    df["margen"] = [0.0] + [exp_total * cart["margen"][esc]] * 3
    return df, exp_total


def main():
    pd.set_option("display.width", 140)
    resultados = {}
    print("PD de S1 que da el satelite (fin 2024 y 2025-2027)")
    for nombre, cart in CARTERAS.items():
        for esc in ESC:
            pd_s, lgd_s = sendas(cart, esc)
            print(f"  {nombre:8} {esc:7}  PD: " + "  ".join(f"{p:6.2%}" for p in pd_s[:4])
                  + "   LGD: " + "  ".join(f"{l:5.1%}" for l in lgd_s[:4]))

    for esc in ESC:
        print(f"\n===== Escenario {esc} =====")
        tot = None
        for nombre in CARTERAS:
            df, exp_total = proyectar(nombre, esc)
            resultados[(nombre, esc)] = (df, exp_total)
            print(f"\n{nombre} (exposicion {exp_total:,.0f} M EUR)")
            print(df[["ano", "S1", "S2", "S3", "prov", "deterioro", "rwa"]].round(1).to_string(index=False))
            acum = df.deterioro.sum()
            print(f"  deterioro acumulado: {acum:,.1f} M EUR = {acum / exp_total:.2%} de la exposicion; "
                  f"S3 final {df.S3.iloc[-1] / exp_total:.2%}; primer ano {df.deterioro.iloc[1] / acum:.0%} del total")
            cols = ["deterioro", "rwa", "margen"]
            tot = df[cols] if tot is None else tot + df[cols]
        rwa0 = tot.rwa.iloc[0]
        for etiqueta, con_margen in (("solo perdidas de credito", False), ("con el margen dado", True)):
            cet1 = CET1_INICIAL_SOBRE_RWA * rwa0
            ratios = [cet1 / rwa0]
            for t in (1, 2, 3):
                resultado = -tot.deterioro.iloc[t] + (tot.margen.iloc[t] if con_margen else 0.0)
                cet1 += resultado * (1 - TIPO_FISCAL) if resultado > 0 else resultado * (1 - TIPO_FISCAL)
                # suelo metodologico: el RWA no puede bajar del de partida (EBA 2027, parr. 153)
                ratios.append(cet1 / max(tot.rwa.iloc[t], rwa0))
            print(f"Filial, CET1 {etiqueta:26}: " + " -> ".join(f"{r:.2%}" for r in ratios)
                  + f"  (caida maxima {(ratios[0] - min(ratios)) * 1e4:,.0f} pb)")
        print(f"RWA de la filial sin suelo: {rwa0:,.0f} -> " + " -> ".join(f"{x:,.0f}" for x in tot.rwa.iloc[1:])
              + f"  (con el suelo, nunca por debajo de {rwa0:,.0f})")

    print("\n===== Adverso menos base =====")
    for nombre in CARTERAS:
        da = resultados[(nombre, "adverso")][0].deterioro.sum()
        db = resultados[(nombre, "base")][0].deterioro.sum()
        e = resultados[(nombre, "base")][1]
        print(f"  {nombre:8} adverso {da / e:.2%}  base {db / e:.2%}  diferencia {(da - db) / e * 100:.2f} pp")

    print("\n===== Prevision perfecta: de donde sale el primer ano (consumo, adverso) =====")
    df_pp, _ = proyectar("consumo", "adverso", True)
    df_np, _ = proyectar("consumo", "adverso", False)
    for etiqueta, df in (("con prevision perfecta", df_pp), ("con la PD del ano siguiente fija", df_np)):
        print(f"  {etiqueta:34} deterioro por ano: "
              + "  ".join(f"{d:6.1f}" for d in df.deterioro.iloc[1:])
              + f"   total {df.deterioro.sum():6.1f}")
    fila = df_pp.iloc[1]
    print(f"  Primer ano, composicion del stock: S1 {fila.prov_S1:.1f}, S2 {fila.prov_S2:.1f}, S3 {fila.prov_S3:.1f}")


if __name__ == "__main__":
    main()


def sensibilidad():
    """Mueve un supuesto cada vez y mide la perdida acumulada de consumo en el adverso."""
    import copy
    original = copy.deepcopy(CARTERAS["consumo"])
    base_df, e = proyectar("consumo", "adverso")
    ref = base_df.deterioro.sum() / e
    print(f"\n===== Sensibilidad (consumo, adverso; referencia {ref:.2%}) =====")
    pruebas = [
        ("LGD 5 puntos menos", "lgd", lambda f: (lambda p, g: f(p, g) - 0.05)),
        ("LGD 5 puntos mas", "lgd", lambda f: (lambda p, g: f(p, g) + 0.05)),
        ("coef. del paro x0,8", "satelite", lambda s: (s[0], s[1] * 0.8, s[2])),
        ("coef. del paro x1,2", "satelite", lambda s: (s[0], s[1] * 1.2, s[2])),
        ("paso a S2 = 1,0 x PD", "m_s1_s2", lambda v: 1.0),
        ("paso a S2 = 2,0 x PD", "m_s1_s2", lambda v: 2.0),
        ("PD de S2 = 2 x PD S1", "m_s2_s3", lambda v: 2.0),
        ("PD de S2 = 4 x PD S1", "m_s2_s3", lambda v: 4.0),
        ("cura de S2 del 10 %", "cura_s2", lambda v: 0.10),
        ("cura de S2 del 30 %", "cura_s2", lambda v: 0.30),
        ("vida residual 2 anos", "vida", lambda v: 2),
        ("vida residual 4 anos", "vida", lambda v: 4),
    ]
    for nombre, clave, cambio in pruebas:
        CARTERAS["consumo"] = copy.deepcopy(original)
        CARTERAS["consumo"][clave] = cambio(original[clave])
        df, e = proyectar("consumo", "adverso")
        v = df.deterioro.sum() / e
        print(f"  {nombre:24} {v:7.2%}  ({(v - ref) * 100:+6.2f} pp)")
    CARTERAS["consumo"] = original


if __name__ == "__main__":
    sensibilidad()

Anexo S1 — Modelo satélite de PD y LGD

Primera parte: estimación de un satélite logit sobre veinte años de datos sintéticos con dos recesiones, lectura de elasticidades, el error de estimar sin crisis y las tres pruebas de validación. Segunda parte: proyección con el escenario de la JERS para México y España, contraste con un benchmark ilustrativo (los del BCE no son públicos) y LGD hipotecaria en función del LTV con la caída de la vivienda del adverso.

"""Anexo S1, primera parte: un modelo satelite estimado sobre datos sinteticos.

Genera veinte anos de datos trimestrales de una cartera de consumo mexicana
(tasa de default observada, paro y crecimiento del PIB), estima un satelite
sobre el logit de la tasa y muestra como se lee: coeficientes, ajuste,
elasticidades y que pasa si se olvida un retardo.

Los datos son sinteticos: el "modelo verdadero" esta escrito abajo y sirve
para comprobar que la estimacion lo recupera.
"""
import numpy as np
import pandas as pd
import statsmodels.api as sm

rng = np.random.default_rng(2026)

# ------------------------------------------------------------------
# 1. Historia macro sintetica: 80 trimestres (20 anos) con dos recesiones
# ------------------------------------------------------------------
T = 80
pib = np.full(T, 2.4)           # crecimiento interanual del PIB, %
pib += rng.normal(0, 0.8, T)
pib[28:34] -= np.array([2.0, 4.5, 6.5, 6.0, 4.0, 1.5])     # recesion 1 (anos 8-9)
pib[60:64] -= np.array([3.0, 9.0, 6.0, 2.0])               # recesion 2 (anos 15-16)

paro = np.empty(T)
paro[0] = 4.0
for t in range(1, T):
    # el paro sube cuando el PIB crece por debajo de ~2,5 % (ley de Okun simplificada)
    paro[t] = paro[t - 1] + 0.12 * (2.5 - pib[t]) * 0.5 - 0.04 * (paro[t - 1] - 4.0)
paro += rng.normal(0, 0.08, T)

# ------------------------------------------------------------------
# 2. Modelo verdadero de la tasa de default anual (en logit)
#    logit(TD_t) = a + b * paro_{t-1} + c * pib_t + ruido
# ------------------------------------------------------------------
#    En recesion la tasa responde mas: termino extra d * max(0, -PIB)
a_true, b_true, c_true, d_true = -3.60, 0.22, -0.06, 0.05
paro_l1 = np.r_[paro[0], paro[:-1]]
logit_td = (a_true + b_true * paro_l1 + c_true * pib + d_true * np.maximum(0, -pib)
            + rng.normal(0, 0.06, T))
td = 1 / (1 + np.exp(-logit_td))

df = pd.DataFrame({"pib": pib, "paro": paro, "paro_l1": paro_l1, "td": td})
df["logit_td"] = np.log(df.td / (1 - df.td))
df = df.iloc[1:]   # se pierde el primer trimestre por el retardo

print("Resumen de los datos sinteticos (80 trimestres)")

"""Anexo S2: migracion entre stages y provisiones con las formulas prescritas de la EBA.

Aplica literalmente los recuadros 3 a 9 de la nota metodologica de 2027 (borrador del
11-jun-2026) a la cartera de consumo de la filial mexicana del hilo conductor (anexo S0):

  Prov S1-S1(t+1) = Exp S1(t) * (1 - TR12(t+1) - PD1(t+1)) * PD1(t+2) * LGD1(t+2)
  Prov S2-S1(t+1) = Exp S2(t) * TR21(t+1) * PD1(t+2) * LGD1(t+2)
  Prov S1-S2(t+1) = Exp S1(t) * TR12(t+1) * LRLT S2(t+1)
  Prov S2-S2(t+1) = Exp S2(t) * (1 - TR21(t+1) - PD2(t+1)) * LRLT S2(t+1)
  Prov S1-S3(t+1) = Exp S1(t) * PD1(t+1) * LGD1(t+1)        (acumuladas, sin liberacion)
  Prov S2-S3(t+1) = Exp S2(t) * PD2(t+1) * LGD2(t+1)
  Prov Old S3     = MAX{Exp S3(0) * LGD S3(1); Prov S3(0)}

En el ultimo ano, los parametros "del ano siguiente" del adverso ponderan 5/6 el adverso y
1/6 el base (reversion lineal en seis anos); en el base quedan planos. LRLT S2 se calcula con
prevision perfecta sobre la vida residual, con la misma reversion tras el horizonte.

Los anos se numeran 1, 2 y 3: el escenario es el de la JERS para Mexico del ejercicio 2025,
y las formulas, las del borrador de 2027.
"""
import numpy as np

PARO_0 = 3.0
ESC = {
    "base":    {"pib": [1.3, 2.0, 2.3],   "paro": [3.3, 3.4, 3.5]},
    "adverso": {"pib": [-4.7, -2.2, -0.1], "paro": [5.2, 7.6, 8.8]},
}
SATELITE = (-3.514, 0.215, -0.0802)      # anexo S1: logit PD = a + b*paro(t-1) + c*PIB(t)
STOCK0 = {"S1": 880.0, "S2": 70.0, "S3": 50.0}
M12, M2, TR21, VIDA = 1.5, 3.0, 0.20, 3    # TR12 = M12*PD1; PD2 = M2*PD1; cura S2->S1; vida residual
EXTRA = 12                                 # anos de sendas tras el horizonte


def logistica(z):
    return 1 / (1 + np.exp(-z))


def lgd(paro):
    return min(0.85, 0.70 + 0.01 * (paro - 3.5))


def sendas(esc):
    """PD1 y LGD para t = 0 (partida), 1-3 (horizonte) y 4.. (tras el horizonte)."""
    a, b, c = SATELITE
    paro_prev = [PARO_0] + ESC[esc]["paro"][:-1]
    pd = [logistica(a + b * PARO_0 + c * ESC["base"]["pib"][0])]
    pd += [logistica(a + b * pp + c * g) for pp, g in zip(paro_prev, ESC[esc]["pib"])]
    lg = [lgd(ESC["base"]["paro"][0])] + [lgd(p) for p in ESC[esc]["paro"]]
    return pd, lg


PD_B, LGD_B = sendas("base")


def extender(pd, lg, esc):
    """Tras el ano 3: base plano; adverso revierte linealmente en 6 anos al base del ano 3."""
    pd, lg = list(pd), list(lg)
    for k in range(1, EXTRA + 1):
        if esc == "adverso":
            w = max(0.0, 1 - k / 6)
            pd.append(PD_B[3] + (pd[3] - PD_B[3]) * w)
            lg.append(LGD_B[3] + (lg[3] - LGD_B[3]) * w)
        else:
            pd.append(pd[3])
            lg.append(lg[3])
    return pd, lg


def lrlt_s2(pd, lg, t, prevision_perfecta=True):
    """Perdida a vida de S2 medida al final del ano t (vida residual VIDA)."""
    sup, perd = 1.0, 0.0
    for j in range(1, VIDA + 1):
        k = t + j if prevision_perfecta else t + 1
        p2 = min(0.6, M2 * pd[k])
        perd += sup * p2 * lg[k]
        sup *= 1 - p2
    return perd


def proyectar(esc, m12=M12, prevision_perfecta=True, reversion_5_6=True, lgd_s3_vieja=None):
    pd, lg = extender(*sendas(esc), esc)
    # parametro "del ano siguiente" en el ano 3: la nota pondera 5/6 adverso y 1/6 base
    def el_sig(t):
        if t + 1 <= 3 or esc == "base":
            return pd[t + 1] * lg[t + 1]
        if reversion_5_6:
            return 5 / 6 * pd[3] * lg[3] + 1 / 6 * PD_B[3] * LGD_B[3]
        return pd[3] * lg[3]

    s1, s2 = STOCK0["S1"], STOCK0["S2"]
    s3_viejo = STOCK0["S3"]
    # partida: se provisiona con el escenario interno del banco (aqui, el base), igual en los dos
    prov_s3_0 = s3_viejo * LGD_B[0]
    prov0 = (s1 * PD_B[1] * LGD_B[1]
             + s2 * lrlt_s2(*extender(PD_B, LGD_B, "base"), 0)
             + prov_s3_0)
    lgd_s3_1 = lg[1] if lgd_s3_vieja is None else lgd_s3_vieja
    prov_viejo = max(s3_viejo * lgd_s3_1, prov_s3_0)            # recuadro 9
    cum_nuevo_s3, prov_cum_s3 = 0.0, 0.0
    filas, prov_ant = [], prov0
    for t in range(0, 3):                                         # de t a t+1
        p1, p2 = pd[t + 1], min(0.6, M2 * pd[t + 1])
        tr12 = m12 * p1
        flujo_13, flujo_23 = s1 * p1, s2 * p2
        flujo_12, flujo_21 = s1 * tr12, s2 * TR21
        prov_13 = flujo_13 * lg[t + 1]
        prov_23 = flujo_23 * lg[t + 1]
        lrlt = lrlt_s2(pd, lg, t + 1, prevision_perfecta)
        prov_11 = s1 * (1 - tr12 - p1) * el_sig(t + 1)
        prov_21 = flujo_21 * el_sig(t + 1)
        prov_12 = flujo_12 * lrlt
        prov_22 = s2 * (1 - TR21 - p2) * lrlt
        s1, s2 = s1 - flujo_13 - flujo_12 + flujo_21, s2 + flujo_12 - flujo_23 - flujo_21
        cum_nuevo_s3 += flujo_13 + flujo_23
        prov_cum_s3 += prov_13 + prov_23
        stock = {"S1": prov_11 + prov_21, "S2": prov_12 + prov_22, "S3": prov_cum_s3 + prov_viejo}
        total = sum(stock.values())
        filas.append({"ano": t + 1, "S1": s1, "S2": s2, "S3": s3_viejo + cum_nuevo_s3,
                      "prov": total, "deterioro": total - prov_ant, "pd1": p1, "lrlt": lrlt,
                      **{f"prov_{k}": v for k, v in stock.items()}})
        prov_ant = total
    return prov0, filas


def resumen(etiqueta, filas, exp=1000.0):

    print(f"  {etiqueta:44} " + " / ".join(f"{d:6.1f}" for d in det)

          + f"  S2 final {filas[-1]['S2']:5.1f}  S3 final {filas[-1]['S3'] / exp:.1%}")
    return sum(det)

Se usa en: capítulo 21. Ejecución: .venv/bin/python -I anexos/s5_reverse.py.

s5_reverse.py

"""Anexo S5: reverse stress test sobre la filial del hilo conductor.

Pregunta inversa: en lugar de "cuanto cae el CET1 con este escenario", "que escenario hace
caer el CET1 tanto". Usa el stress test de juguete del anexo S0 (filial mexicana, consumo y
empresas, margen dado) y dos herramientas:

1. Escalar el escenario: el adverso se escribe como base + k * (adverso - base) y se busca la
   k que produce la caida objetivo. k = 1 es el adverso de la JERS.
2. El escenario mas plausible: se mueven por separado el choque de paro y el de PIB, se
   trazan las combinaciones que producen la misma caida y se elige la que esta mas cerca del
   base en distancia de Mahalanobis (desviaciones tipicas y correlacion supuestas, ilustrativas).
"""
import sys
from pathlib import Path

import numpy as np
from scipy.optimize import brentq

sys.path.insert(0, str(Path(__file__).resolve().parent))
import s0_juguete as s0  # noqa: E402

BASE = {k: list(v) for k, v in s0.ESC["base"].items()}
ADV = {k: list(v) for k, v in s0.ESC["adverso"].items()}


def caida_maxima(k_paro, k_pib):
    """Caida maxima de CET1 (pb) de la filial con el adverso escalado, margen adverso dado."""
    s0.ESC["adverso"] = {
        "paro": [b + k_paro * (a - b) for a, b in zip(ADV["paro"], BASE["paro"])],
        "pib": [b + k_pib * (a - b) for a, b in zip(ADV["pib"], BASE["pib"])],
    }
    tot = None
    for nombre in s0.CARTERAS:
        df, _ = s0.proyectar(nombre, "adverso")
        cols = df[["deterioro", "rwa", "margen"]]
        tot = cols if tot is None else tot + cols
    rwa0 = tot.rwa.iloc[0]
    cet1 = s0.CET1_INICIAL_SOBRE_RWA * rwa0
    ratios = [cet1 / rwa0]
    for t in (1, 2, 3):
        cet1 += (-tot.deterioro.iloc[t] + tot.margen.iloc[t]) * (1 - s0.TIPO_FISCAL)
        ratios.append(cet1 / max(tot.rwa.iloc[t], rwa0))
    s0.ESC["adverso"] = {k: list(v) for k, v in ADV.items()}
    return (ratios[0] - min(ratios)) * 1e4


def main():
    ref = caida_maxima(1.0, 1.0)
    print(f"Adverso JERS (k = 1): caida maxima {ref:.0f} pb")
    print("\n1. Escalar el escenario entero")
    for objetivo in (300, 600, 800):
        k = brentq(lambda x: caida_maxima(x, x) - objetivo, 0.0, 3.0)
        paro3 = BASE["paro"][2] + k * (ADV["paro"][2] - BASE["paro"][2])
        pib1 = BASE["pib"][0] + k * (ADV["pib"][0] - BASE["pib"][0])
        print(f"  Caida de {objetivo} pb: k = {k:.2f} (paro del ano 3 {paro3:.1f} %, PIB del ano 1 {pib1:.1f} %)")

    print("\n2. El escenario mas plausible para una caida de 600 pb")
    # desviaciones tipicas del choque (en unidades de k) y correlacion, supuestas
    sd_paro, sd_pib, rho = 1.0, 1.0, 0.6
    cov = np.array([[sd_paro ** 2, rho * sd_paro * sd_pib], [rho * sd_paro * sd_pib, sd_pib ** 2]])
    inv = np.linalg.inv(cov)
    mejor = None
    for kp in np.linspace(0.0, 3.0, 61):
        f = lambda kg: caida_maxima(kp, kg) - 600  # noqa: E731
        if f(-1.0) * f(6.0) > 0:
            continue
        kg = brentq(f, -1.0, 6.0)
        d = float(np.sqrt(np.array([kp, kg]) @ inv @ np.array([kp, kg])))
        if mejor is None or d < mejor[2]:
            mejor = (kp, kg, d)
        if abs(kp - round(kp * 2) / 2) < 1e-9:
            print(f"  choque de paro x{kp:.1f} -> hace falta choque de PIB x{kg:.2f}  (distancia {d:.2f})")
    kp, kg, d = mejor
    print(f"  Mas plausible: paro x{kp:.2f}, PIB x{kg:.2f} (distancia {d:.2f}); "
          f"escalado uniforme: x{brentq(lambda x: caida_maxima(x, x) - 600, 0, 3):.2f}")


if __name__ == "__main__":
    main()

"""Anexo S6: la misma cartera en tres ejercicios, y el puente entre ellos.

La cartera de consumo de la filial mexicana del hilo conductor (1.000 M EUR: 880 / 70 / 50)
se proyecta tres veces con la misma maquinaria (satelite del anexo S1, formulas de provisiones
del anexo S2) y supuestos distintos:

  1. Europeo (EBA): adverso JERS (k = 1), balance estatico, sin acciones de gestion.
  2. ICAAP: escenario propio mas severo (k = 1,15), balance dinamico: vence un 30 % de lo sano
     cada ano y se repone con produccion nueva mas selectiva (PD x 0,85), sin crecimiento.
  3. Local tipo supervisor mexicano: escenario del supervisor (k = 0,90), plan de negocio con
     crecimiento del 8 % anual de la cartera sana; misma admision que hoy.

k escala la desviacion del adverso sobre el base (paro y PIB). Todos los supuestos de los
ejercicios 2 y 3 son ilustrativos: los escenarios locales y los planes reales no son publicos.
El puente descompone la diferencia con el europeo en efecto escenario y efecto balance.
"""
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent))
import s2_migracion as s2  # noqa: E402

BASE = {k: list(v) for k, v in s2.ESC["base"].items()}
ADV = {k: list(v) for k, v in s2.ESC["adverso"].items()}


def sendas_k(k):
    s2.ESC["adverso"] = {v: [b + k * (a - b) for a, b in zip(ADV[v], BASE[v])] for v in ADV}
    pd, lg = s2.extender(*s2.sendas("adverso"), "adverso")
    s2.ESC["adverso"] = {v: list(x) for v, x in ADV.items()}
    return pd, lg


def proyectar(k, vence=0.0, crece=0.0, m_nueva=1.0):
    """Perdida acumulada (deterioro de 3 anos) y exposicion media; formulas del anexo S2."""
    pd, lg = sendas_k(k)
    s1v, s1n, s2_ = s2.STOCK0["S1"], 0.0, s2.STOCK0["S2"]
    s3_viejo = s2.STOCK0["S3"]
    prov_s3_0 = s3_viejo * s2.LGD_B[0]
    prov_ant = (s1v * s2.PD_B[1] * s2.LGD_B[1]
                + s2_ * s2.lrlt_s2(*s2.extender(s2.PD_B, s2.LGD_B, "base"), 0) + prov_s3_0)
    prov_viejo = max(s3_viejo * lg[1], prov_s3_0)
    prov_cum_s3, det_total, expos = 0.0, 0.0, []
    for t in range(3):
        sig = lambda m: m * (pd[t + 2] * lg[t + 2] if t < 2 else  # noqa: E731
                             5 / 6 * pd[3] * lg[3] + 1 / 6 * s2.PD_B[3] * s2.LGD_B[3])
        nuevos = 0.0
        for cohorte, m in (("v", 1.0), ("n", m_nueva)):
            stock = s1v if cohorte == "v" else s1n
            p1 = m * pd[t + 1]
            f13, f12 = stock * p1, stock * s2.M12 * p1
            prov_cum_s3 += f13 * lg[t + 1]
            nuevos += f12
            if cohorte == "v":
                s1v = stock - f13 - f12
            else:
                s1n = stock - f13 - f12
        p2 = min(0.6, s2.M2 * pd[t + 1])
        f23, f21 = s2_ * p2, s2_ * s2.TR21
        prov_cum_s3 += f23 * lg[t + 1]
        s2_ = s2_ - f23 - f21 + nuevos
        s1v += f21
        # vencimientos y produccion nueva (balance dinamico)
        sano_ini = s1v + s1n + s2_
        s1v, s1n, s2_ = (x * (1 - vence) for x in (s1v, s1n, s2_))
        objetivo = (s2.STOCK0["S1"] + s2.STOCK0["S2"]) * (1 + crece) ** (t + 1)
        if vence > 0 or crece > 0:
            s1n += max(0.0, objetivo - (s1v + s1n + s2_)) if crece > 0 else sano_ini * vence
        prov = ((s1v * sig(1.0) + s1n * sig(m_nueva))
                + s2_ * s2.lrlt_s2(pd, lg, t + 1) + prov_cum_s3 + prov_viejo)
        det_total += prov - prov_ant
        prov_ant = prov
        expos.append(s1v + s1n + s2_)
    return det_total, sum(expos) / 3


CASOS = {
    "europeo": dict(k=1.00),
    "icaap": dict(k=1.15, vence=0.30, m_nueva=0.85),
    "local": dict(k=0.90, vence=0.30, crece=0.08),
}


def main():
    res = {}
    for nombre, kw in CASOS.items():
        det, media = proyectar(**kw)
        res[nombre] = det
        print(f"{nombre:8} deterioro 3 anos {det:6.1f} M EUR = {det / 1000:.2%} de la exposicion inicial; "
              f"exposicion sana media {media:6.1f}")
    print("\nPuente con el europeo (puntos de perdida sobre la exposicion inicial):")
    for nombre in ("icaap", "local"):
        kw = CASOS[nombre]
        solo_esc, _ = proyectar(k=kw["k"])
        e_esc = (solo_esc - res["europeo"]) / 10
        e_bal = (res[nombre] - solo_esc) / 10
        print(f"  {nombre:6}: europeo {res['europeo'] / 10:5.2f} + escenario {e_esc:+5.2f} + balance {e_bal:+5.2f}"
              f" = {res[nombre] / 10:5.2f}")


if __name__ == "__main__":
    main()
